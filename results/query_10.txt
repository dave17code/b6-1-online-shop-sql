Source: sql/03_queries.sql
Order: Q01 to Q15 from the original sample data

[SQL]
-- Use the historical unit_price, not the current product price.
SELECT product_name, SUM(quantity) AS total_quantity,
       SUM(quantity * unit_price) AS total_amount
FROM products
INNER JOIN order_items USING (product_id)
GROUP BY product_id, product_name
ORDER BY total_amount DESC, product_id;
